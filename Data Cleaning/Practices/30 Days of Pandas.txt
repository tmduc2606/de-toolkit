import pandas as pd

# pd.merge(left, right, how='inner', on=None, left_on=None, right_on=None, left_index=False, right_index=False, sort=False, suffixes=('_x', '_y'), copy=<no_default>, indicator=False, validate=None)

def find_customers(customers: pd.DataFrame, orders: pd.DataFrame) -> pd.DataFrame:
    # merge() = SQL-like JOINS with how = left, right, inner...
    # left_on = fixed column to be joined
    # right_on = column to join on the right side
    result = pd.merge(
        customers, 
        orders, 
        how = "left", 
        left_on = "id", 
        right_on = "customerId"
    )

    # Filter all rows with NaN values in Orders.customerId
    result = result[pd.isna(result["customerId"])]

    # Must use nested list <=> DataFrame (otherwise Series)
    return result[["name"]].rename(columns = {"name": "Customers"})

1873. Calculate Special Bonus

import pandas as pd

# This method will not work -> Series-wise
#     if (employees["employee_id"] % 2 != 0) & (employees["name"].str.startswith("M")):

# Use indexing instead
# loc[]
# Syntax: DataFrame.loc 
# Parameter : None 
# Returns : Scalar, Series, DataFrame

def calculate_special_bonus(employees: pd.DataFrame) -> pd.DataFrame:
    
    # By default, employee doesn't have any bonus
    employees["bonus"] = 0

    # ID % 2 != 0 and emp name does not start with "M"
    mask = (employees["employee_id"] % 2 != 0) & (~employees["name"].str.startswith("M"))
    
    # Apply 100% bonus (salary) on conditional employees
    employees.loc[mask, "bonus"] = employees.loc[mask, "salary"]
    
    return employees[["employee_id", "bonus"]].sort_values(by = "employee_id", ascending = True)

1517. Find Users With Valid E-mails

import pandas as pd

# [A-Za-z]+ -> (One or more) The first letter must be a char
# [A-Za-z0-9._-]* -> letters (upper, lower), digits, underscore, period and dash
# fullmatch() -> entire match a regex

def valid_emails(users: pd.DataFrame) -> pd.DataFrame:
    pattern = r'[A-Za-z]+[A-Za-z0-9._-]*@leetcode\.com'
    return users[users["mail"].str.fullmatch(pattern)]

1527. Patients With a Condition
# (^|.*\s) - ^ matches the start of the string,
# . = any character
# * = zero or more character
# \s = a whitespace character (space, tab, newline)
# Sum in: matches everything up to a space

def find_patients(patients: pd.DataFrame) -> pd.DataFrame:
    return patients[patients["conditions"].str.match(r"(^|.*\s)DIAB1", na = False)]

177. Nth Highest Salary
import pandas as pd

def nth_highest_salary(employee: pd.DataFrame, N: int) -> pd.DataFrame:

    # Return result of second, third, ... largest salaries
    result = employee.drop_duplicates(subset = ["salary"]).nlargest(N, "salary")

    # N invalid -> null or,
    # If the result is less than distinct salaries, not enough
    # for leaderboard -> null
    if (N <= 0) or (len(result) < N):
        return pd.DataFrame({f"getNthHighestSalary({N})": [None]})

    # Return the result of the last row (least top)
    return result[["salary"]].iloc[[-1]].rename(columns = {"salary": f"getNthHighestSalary({N})"})

176. Second Highest Salary

import pandas as pd

def second_highest_salary(employee: pd.DataFrame) -> pd.DataFrame:
    
    # If the leaderboard is singular
    if len(pd.unique(employee["salary"])) <= 1:
        return pd.DataFrame({"SecondHighestSalary": [None]})

    # drop_duplicates() -> the leaderboard may contain duplicated values on salary
    # .nlargest(2, "salary") -> return the second, third, ... highest salaries
    # [["salary"]].iloc[[-1]] -> return the non-highest leaderboard, then select the last row
    # .rename(columns = {"salary": "SecondHighestSalary"}) -> rename the column to SecondHighestSalary
    return employee.drop_duplicates(subset = ["salary"]).nlargest(2, "salary")[["salary"]].iloc[[-1]].rename(columns = {"salary": "SecondHighestSalary"})

184. Department Highest Salary

import pandas as pd

# df.rank(axis=0, method='average', numeric_only=False, na_option='keep', ascending=True, pct=False)

def department_highest_salary(employee: pd.DataFrame, department: pd.DataFrame) -> pd.DataFrame:
    
    # Rename the columns for neat return
    employee = employee.rename(columns = {"name": "Employee", "salary": "Salary"})
    department = department.rename(columns = {"name": "Department"})

    # SQL-like JOIN on department ID
    merged = pd.merge(employee, department, left_on = "departmentId", right_on = "id")

    # RANK() OVER (PARTITION BY departmentId ORDER BY Salary DESC)
    merged["rank"] = merged.groupby("departmentId")["Salary"].rank(method = "min", ascending = False)

    # LIMIT 1
    result = merged[merged["rank"] == 1]

    # Return respective columns
    return result[["Department", "Employee", "Salary"]]

1907. Count Salary Categories

import pandas as pd

# df.apply(func, axis=0, raw=False, result_type=None, args=(), by_row='compat', engine=None, engine_kwargs=None, **kwargs) -> apply to either column-wise or row-wise

def count_salary_categories(accounts: pd.DataFrame) -> pd.DataFrame:
    
    # Classify function
    def classify(income):
        if income < 20000:
            return "Low Salary"
        if (income >= 20000) and (income <= 50000):
            return "Average Salary"
        return "High Salary"
    
    # Apply to the cateogry feature
    accounts["category"] = accounts["income"].apply(classify)

    # Count the total number of cateogry
    counts = accounts.groupby("category").size()

    # Construct the dataframe with intial category
    result = pd.DataFrame({"category": ["Low Salary", "Average Salary", "High Salary"]}).sort_values("category", ascending = True)

    # Map the counts with the category
    result["accounts_count"] = result["category"].map(counts).fillna(0) # .astype(int) for safer casting

    return result

1741. Find Total Time Spent by Each Employee

import pandas as pd

# pd.groupby(by=None, level=None, *, as_index=True, sort=True, group_keys=True, observed=True, dropna=True)

# as_index - Return object with group labels as the index. Only relevant for DataFrame input. as_index=False is effectively “SQL-style” grouped output.

def total_time(employees: pd.DataFrame) -> pd.DataFrame:

    # Time spent in the office for a single entry
    employees["total_time"] = employees["out_time"] - employees["in_time"]

    # Group by event_day, emp_id
    result = employees.groupby(["event_day", "emp_id"], as_index = False)["total_time"].sum()

    return result.rename(columns = {"event_day": "day"})

607. Sales Person

import pandas as pd

def sales_person(sales_person: pd.DataFrame, company: pd.DataFrame, orders: pd.DataFrame) -> pd.DataFrame:

    # Return all sales_id whose order belongs to "RED" company
    sales = orders.merge(company, on = "com_id").loc[lambda df: df["name"].isin(["RED"]), "sales_id"]

    # Filters those who don't have orders from RED company (negate)
    return sales_person[~sales_person["sales_id"].isin(sales)][["name"]]

570. Managers with at Least 5 Direct Reports

import pandas as pd

def find_managers(employee: pd.DataFrame) -> pd.DataFrame:
    result = employee.groupby("managerId")["id"].size().reset_index(name = "count")
    
    # Join the result (>= 5 direct reports) with the origin (equivalent to self-join SQL)
    return result[result["count"] >= 5].merge(employee, left_on = "managerId", right_on = "id")[["name"]]

1280. Students & Examinations

import pandas as pd

def students_and_examinations(students: pd.DataFrame, subjects: pd.DataFrame, examinations: pd.DataFrame) -> pd.DataFrame:

    # A default table where all students learn 3 subjects
    table = students.merge(subjects, how = "cross")

    # Count of attended exams in examinations table
    result = examinations.groupby(["student_id", "subject_name"]).size().reset_index(name = "attended_exams")

    # Merge the default table with the result (left-join) -> fill null values with 0 -> rearrange the columns
    # Order by student_id, subject_name
    return table.merge(result, on = ["student_id", "subject_name"], how = "left").fillna({"attended_exams": 0})[["student_id", "student_name", "subject_name", "attended_exams"]].sort_values(by = ["student_id", "subject_name"], ascending = True)
